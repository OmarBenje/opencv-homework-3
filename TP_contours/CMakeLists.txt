cmake_minimum_required(VERSION 3.16)
project(TP_contours VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
add_library(contours_core src/contours.cpp)
target_include_directories(contours_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(contours_core PUBLIC ${OpenCV_LIBS})
add_executable(contours src/main.cpp)
target_link_libraries(contours PRIVATE contours_core)
add_executable(contours_tests tests/test_contours.cpp)
target_link_libraries(contours_tests PRIVATE contours_core)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(contours_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(contours PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(contours_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME algorithms_and_matching COMMAND contours_tests)
add_test(NAME cli_help COMMAND contours --help)
add_test(NAME reject_bad_detector COMMAND contours --detector unknown)
set_tests_properties(reject_bad_detector PROPERTIES WILL_FAIL TRUE)
